> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `error-triage-service` the single backend for the "Report a problem" panel by adding the four behaviors experttech built in September: idempotent submissions, a human ticket reference, service-owned attachments surfaced on the Plane issue, and a reporter confirmation email.

**Architecture:** All changes land in `~/projects/error-triage-service` (Next.js 14 App Router, Supabase service-role client, vitest with a hand-rolled supabase mock). Two new migrations, one new route (`/api/v1/attachments/sign`), two small new libs (`lib/attachments/*`, `lib/email/reporter-confirmation.ts`), and additions to `app/api/v1/report/route.ts`. Attachment bytes go browser to Supabase storage directly via signed upload URLs; the service only issues tokens and later signs read URLs.

**Tech Stack:** Next.js 14.2, `@supabase/supabase-js` 2.47, zod 3, vitest 2, Postgres via `supabase/migrations`.

**Spec:** `~/projects/report-a-problem/docs/superpowers/specs/2026-09-05-consolidation-design.md` (section "Service changes").

```ts
// The reporter-facing reference ("REP-1042"). Ported from experttech lib/errors/ticket.ts,

export function formatTicketRef(prefix: string, n: number | null | undefined): string | null {
```

```sql
-- 20260905000010_report_attachments.sql
-- Service-owned attachments. Rows are created when an upload is SIGNED (bound to
-- app + reporter), and claimed by a report later. Ownership at report time is a
-- keyed lookup on (id, app_id, reporter_id, report_id IS NULL), never a path
-- prefix test.
CREATE TABLE public.report_attachments (
  id          uuid PRIMARY KEY,
  app_id      uuid NOT NULL REFERENCES public.apps(id) ON DELETE CASCADE,
  reporter_id text NOT NULL,
  tenant_id   text,
  path        text NOT NULL UNIQUE,
  mime        text NOT NULL,
  size_bytes  integer NOT NULL CHECK (size_bytes > 0 AND size_bytes <= 8388608),
  report_id   uuid REFERENCES public.error_reports(id) ON DELETE SET NULL,
  created_at  timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX idx_report_attachments_report ON public.report_attachments (report_id);
CREATE INDEX idx_report_attachments_unclaimed ON public.report_attachments (created_at) WHERE report_id IS NULL;
ALTER TABLE public.report_attachments ENABLE ROW LEVEL SECURITY;
REVOKE ALL ON public.report_attachments FROM anon, authenticated;

ALTER TABLE public.error_reports
  ADD COLUMN attachment_ids uuid[] NOT NULL DEFAULT '{}';

INSERT INTO storage.buckets (id, name, public, file_size_limit, allowed_mime_types)

    'image/png', 'image/jpeg', 'image/webp',

    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'

)
ON CONFLICT (id) DO NOTHING;
```

```ts
// app/api/v1/attachments/sign/route.ts
//
// POST /api/v1/attachments/sign — issue a one-shot signed upload URL for a report
// attachment. The browser PUTs bytes straight to storage; no attachment bytes
// pass through the app forwarder or this function (Vercel caps function bodies
// at 4.5MB). The row is created NOW, bound to (app, reporter), so the later
// report can claim it by id rather than trusting a client-supplied path.
import { randomUUID } from 'node:crypto';
import { NextRequest, NextResponse } from 'next/server';
import { z } from 'zod';
import { createServiceRoleClient } from '@/lib/supabase/server';
import { getOrCreateRequestId, logEvent } from '@/lib/observability/logger';
import { resolveAppByKey } from '@/lib/registry/app-auth';
import { decideGate, type TenantGating } from '@/lib/registry/gate';
import { scrub, sanitizeId } from '@/lib/errors/scrub';
import { rateLimited } from '@/lib/errors/rate-limit';
import { sanitizeAttachmentFilename } from '@/lib/attachments/filename';
import { ATTACHMENT_ALLOWED_MIME, ATTACHMENT_BUCKET, ATTACHMENT_MAX_BYTES } from '@/lib/attachments/policy';

export const maxDuration = 15;

const BodySchema = z.object({
  reporter: z.object({ id: z.string().min(1) }),
  tenant_id: z.string().nullish(),
  filename: z.string().nullish(),
  mime: z.string().min(1),
  size_bytes: z.number().int(),
});

function clientIp(req: NextRequest): string {
  return (req.headers.get('x-forwarded-for') ?? '').split(',')[0].trim() || 'unknown';
}

export async function POST(request: NextRequest) {
  const requestId = getOrCreateRequestId(request.headers.get('x-request-id'));
  const reply = (status: number, payload: Record<string, unknown>) =>
    NextResponse.json(payload, { status, headers: { 'x-request-id': requestId } });
  try {
    const supabase = createServiceRoleClient();
    const app = await resolveAppByKey(supabase, request.headers.get('x-triage-key'));
    if (!app) return reply(401, { ok: false, error: 'unauthorized' });

    let json: unknown;
    try { json = await request.json(); } catch { return reply(400, { ok: false, error: 'invalid_json' }); }
    const parsed = BodySchema.safeParse(json);
    if (!parsed.success) return reply(400, { ok: false, error: 'invalid_body' });
    const b = parsed.data;

    const reporterId = scrub(b.reporter.id) ?? '';
    if (!reporterId) return reply(400, { ok: false, error: 'invalid_body' });
    const tenantId = scrub(b.tenant_id?.trim() || undefined) ?? null;
    if (b.size_bytes <= 0) return reply(400, { ok: false, error: 'invalid_body' });
    if (b.size_bytes > ATTACHMENT_MAX_BYTES) return reply(413, { ok: false, error: 'too_large' });
    if (!ATTACHMENT_ALLOWED_MIME.has(b.mime)) return reply(415, { ok: false, error: 'unsupported_type' });

    if (await rateLimited(supabase, `${app.keyId}:attach:${tenantId ?? 'none'}:${clientIp(request)}`)) {
      return reply(429, { ok: false, error: 'rate_limited' });
    }

    const { data: appRow } = await supabase.from('apps').select('enabled, tenant_gating').eq('id', app.appId).maybeSingle();
    if (!appRow) return reply(401, { ok: false, error: 'unauthorized' });
    const cfg = appRow as { enabled: boolean; tenant_gating: TenantGating };
    let tenantConfig: boolean | null = null;
    if (cfg.tenant_gating !== 'all' && tenantId) {
      const { data: tc } = await supabase.from('app_tenant_config').select('enabled').eq('app_id', app.appId).eq('tenant_id', tenantId).maybeSingle();
      tenantConfig = tc ? (tc as { enabled: boolean }).enabled : null;
    }
    const gate = decideGate({ appEnabled: cfg.enabled, tenantGating: cfg.tenant_gating, tenantId, tenantConfig });
    if (gate.decision !== 'file') return reply(403, { ok: false, error: 'gated' });

    const id = randomUUID();
    const filename = sanitizeAttachmentFilename(sanitizeId(b.filename ?? undefined), b.mime);
    const path = `${app.appId}/${id}-${filename}`;

    const { error: insErr } = await supabase.from('report_attachments').insert({
      id, app_id: app.appId, reporter_id: reporterId, tenant_id: tenantId, path, mime: b.mime, size_bytes: b.size_bytes,
    });
    if (insErr) {
      logEvent('error', 'attachment.sign_insert_failed', { requestId, appId: app.appId, error: insErr.message });
      return reply(503, { ok: false, error: 'db_unavailable' });
    }

    const { data: signedData, error: signErr } = await supabase.storage.from(ATTACHMENT_BUCKET).createSignedUploadUrl(path);
    if (signErr || !signedData?.signedUrl) {
      logEvent('error', 'attachment.sign_failed', { requestId, appId: app.appId, error: signErr?.message });
      return reply(503, { ok: false, error: 'storage_unavailable' });
    }
    logEvent('info', 'attachment.signed', { requestId, appId: app.appId, attachmentId: id, bytes: b.size_bytes });
    return reply(200, { ok: true, attachment_id: id, upload_url: signedData.signedUrl });
  } catch (e) {
    logEvent('error', 'attachment.sign_handler_failed', { requestId, error: e instanceof Error ? e.message : 'unknown' });
    return reply(500, { ok: false, error: 'sign_failed' });
  }
}
```

`decideGate` returns `decision` as one of `'file' | 'noop' | 'reject'` (verified in `lib/registry/gate.ts`); only `'file'` may sign.

```ts
// lib/email/reporter-confirmation.ts
// Confirmation copy sent to the REPORTER (not the team) for a manual report.
// Generic across apps: the app's display_name is the only branding.
function escapeHtml(s: string): string {
  return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}
export function renderReporterConfirmation(args: {
  name: string | null; ref: string; description: string; route: string | null; appName: string; timestamp: string;
}): { subject: string; html: string } {
  const first = (args.name ?? '').split(' ')[0] || 'there';
  const desc = escapeHtml(args.description).replace(/\n/g, '<br>');
  return {
    subject: `We've got your report (${args.ref})`,
    html: `<div style="font-family:system-ui,sans-serif;max-width:560px;margin:0 auto;padding:24px;color:#111827">

<table width="100%" cellpadding="0" cellspacing="0" style="margin:16px 0 24px;background:#f9fafb;border-radius:6px;border:1px solid #e5e7eb"><tr><td style="padding:16px 20px">
```
